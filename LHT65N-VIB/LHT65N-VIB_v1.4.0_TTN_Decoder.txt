function datalog(i,bytes){
  var aa= parseFloat((bytes[i]<<24>>16 | bytes[i+1])/1000).toFixed(3); 
  var bb= parseFloat((bytes[i+2]<<24>>16 | bytes[i+3])/1000).toFixed(3); 
  var cc= parseFloat((bytes[i+4]<<24>>16 | bytes[i+5])/1000).toFixed(3); 
  var string='[('+ aa +'),'+'('+ bb +'),' +'('+ cc +')]'+',';  
  return string;
}

function getAxisStateTransitionStringByByte(bytes) {
  const map = {
    0x01: "LowToLow",
    0x02: "LowToNormal",
    0x03: "LowToHigh",
    0x04: "NormalToLow",
    0x05: "NormalToNormal",
    0x06: "NormalToHigh",
    0x07: "HighToLow",
    0x08: "HighToNormal",
    0x09: "HighToHigh"
  };
  return map[bytes] ?? "AxisStateTransitionStringNotFound";
}

function formatTimestamp(timestamp, unit, separator) {
    if (unit === undefined) {
        unit = 's';
    }
    if (separator === undefined) {
        separator = ' ';
    }
    
    var ts = Number(timestamp);
    
    if (isNaN(ts)) {
        throw new Error('Invalid timestamp: Must be a number');
    }

    if (unit === 's') ts *= 1000;
    else if (unit !== 'ms') throw new Error('Invalid unit');
    
    const date = new Date(ts);

    if (isNaN(date.getTime())) {
        throw new Error('Invalid timestamp: Cannot be converted to a valid date');
    }
    
    var year = date.getFullYear();
    var month = String(date.getMonth() + 1).padStart(2, '0');
    var day = String(date.getDate()).padStart(2, '0');
    var hours = String(date.getHours()).padStart(2, '0');
    var minutes = String(date.getMinutes()).padStart(2, '0');
    var seconds = String(date.getSeconds()).padStart(2, '0');

    return year + '-' + month + '-' + day + separator + hours + ':' + minutes + ':' + seconds;
}

function Decoder(bytes, port) {
  var decode = {};
  if(port==0x02)
  {
    decode.BatV=(bytes[0]<<8 | bytes[1])/1000;
    var mod=(bytes[2]>>2)&0x07;
    if(mod==1)
    {
      decode.vib_count=(bytes[3]<<8 | bytes[4]<<8 | bytes[5]<<8 | bytes[6])>>>0;
      decode.work_min=(bytes[7]<<8 | bytes[8]<<8 | bytes[9]<<8 | bytes[10])>>>0;
    }
    else if(mod==2)
    {
      decode.vib_count=(bytes[3]<<8 | bytes[4]<<8 | bytes[5]<<8 | bytes[6])>>>0;    
      decode.TempC_SHT=parseFloat(((bytes[7]<<24>>16 | bytes[8])/100).toFixed(2));
      decode.Hum_SHT=parseFloat((((bytes[9]<<8 | bytes[10])&0xFFF)/10).toFixed(1));
    }
    else if(mod==3)
    {
      decode.TempC_SHT=parseFloat(((bytes[3]<<24>>16 | bytes[4])/100).toFixed(2));
      decode.Hum_SHT=parseFloat((((bytes[5]<<8 | bytes[6])&0xFFF)/10).toFixed(1));
      decode.work_min=(bytes[7]<<8 | bytes[8]<<8 | bytes[9]<<8 | bytes[10])>>>0;
    }

    decode.Alarm= (bytes[2] & 0x01)? "TRUE":"FALSE";
    decode.TDC  = (bytes[2] & 0x02)? "YES":"NO";
	decode.Node_type="LHT65N-VIB";
    
    if(bytes.length==11)
      return decode;
  }
  else if(port==7)
  {
    var Bat= (bytes[0]<<8 | bytes[1])/1000;
    for(var k=2;k<bytes.length;k=k+6)
    {
      data= datalog(k,bytes);
      if(k=='2')
        data_sum=data;
      else
        data_sum+=data;
    }
    return{
	Node_type:"LHT65N-VIB",
    Bat_V:Bat,
    DATALOG:data_sum
    };    
  }  
  else if (port === 9) {
    decode.BatV = (bytes[0] << 8 | bytes[1]) / 1000;
    
    decode.max_acc_x_g = parseFloat((bytes[2] << 24 >> 16 | bytes[3]) / 1000).toFixed(3);
    decode.max_acc_y_g = parseFloat((bytes[4] << 24 >> 16 | bytes[5]) / 1000).toFixed(3);
    decode.max_acc_z_g = parseFloat((bytes[6] << 24 >> 16 | bytes[7]) / 1000).toFixed(3);
    
    decode.Node_type = "LHT65N-VIB";

    if (bytes.length === 8)
      return decode;
  }
  else if (port === 10) {
    decode.BatV = (bytes[0] << 8 | bytes[1]) / 1000;
    
    decode.acc_x_g = parseFloat((bytes[2] << 24 >> 16 | bytes[3]) / 1000).toFixed(3);
    decode.acc_y_g = parseFloat((bytes[4] << 24 >> 16 | bytes[5]) / 1000).toFixed(3);
    decode.acc_z_g = parseFloat((bytes[6] << 24 >> 16 | bytes[7]) / 1000).toFixed(3);
    
    if (bytes[8] & 0x80) {
      decode.X_state_chg = getAxisStateTransitionStringByByte(bytes[8] & (~0x80));
    } else {
      decode.x_state_chg = getAxisStateTransitionStringByByte(bytes[8] & (~0x80));
    }

    if (bytes[9] & 0x80) {
      decode.Y_state_chg = getAxisStateTransitionStringByByte(bytes[9] & (~0x80));
    } else {
      decode.y_state_chg = getAxisStateTransitionStringByByte(bytes[9] & (~0x80));
    }
    
    if (bytes[10] & 0x80) {
      decode.Z_state_chg = getAxisStateTransitionStringByByte(bytes[10] & (~0x80));
    } else {
      decode.z_state_chg = getAxisStateTransitionStringByByte(bytes[10] & (~0x80));
    }
    
    decode.Node_type = "LHT65N-VIB";
    
    if (bytes.length === 11)
      return decode;
  }
  else if (port === 12) {
    if (bytes[0] === 0x45) {
      if (bytes[1] === 0x05) {
        if (bytes[2] === 0x00) {
          decode.odr = getOdrStringByByte(bytes[3]);
          decode.op_md = getOpmdStringByByte(bytes[4]);
          decode.fs = getFsStringByByte(bytes[5]);
          decode.watermark = bytes[6];
          decode.tdc = bytes[7] << 24 >> 16 | bytes[8];
          
          decode.Node_type = "LHT65N-VIB";

          if (bytes.length === 9)
            return decode;
        }
      } else if (bytes[1] === 0x06) {
        if (bytes[2] === 0x00) {
          decode.odr = getOdrStringByByte(bytes[3]);
          decode.op_md = getOpmdStringByByte(bytes[4]);
          decode.fs = getFsStringByByte(bytes[5]);
          decode.watermark = bytes[6];
          decode.keepalive_interval_s = bytes[7] << 24 >> 16 | bytes[8];
          decode.queue_capacity = bytes[9];
          
          decode.Node_type = "LHT65N-VIB";

          if (bytes.length === 10)
            return decode;
        } else if (bytes[2] === 0x01) {
          decode.x_axis_low_thld = bytes[3] << 24 >> 16 | bytes[4];
          decode.x_axis_high_thld = bytes[5] << 24 >> 16 | bytes[6];
          decode.x_axis_low_thld_hyst_half_width = bytes[7] << 24 >> 16 | bytes[8];
          decode.x_axis_high_thld_hyst_half_width = bytes[9] << 24 >> 16 | bytes[10];

          decode.Node_type = "LHT65N-VIB";

          if (bytes.length === 11)
            return decode;
        } else if (bytes[2] === 0x02) {

          decode.y_axis_high_thld = bytes[5] << 24 >> 16 | bytes[6];
          decode.y_axis_low_thld_hyst_half_width = bytes[7] << 24 >> 16 | bytes[8];
          decode.y_axis_high_thld_hyst_half_width = bytes[9] << 24 >> 16 | bytes[10];

          if (bytes.length === 11)
            return decode;
        } else if (bytes[2] === 0x03) {
          decode.z_axis_low_thld = bytes[3] << 24 >> 16 | bytes[4];
          decode.z_axis_high_thld = bytes[5] << 24 >> 16 | bytes[6];
          decode.z_axis_low_thld_hyst_half_width = bytes[7] << 24 >> 16 | bytes[8];
          decode.z_axis_high_thld_hyst_half_width = bytes[9] << 24 >> 16 | bytes[10];

          decode.Node_type = "LHT65N-VIB";
          
          if (bytes.length === 11)
            return decode;
        } else if (bytes[2] === 0x04) {
          decode.thld_state_update_method = getThldStateUpdateMethodStringByByte(bytes[3]);
          decode.trigger_filter = getTriggerFilterStringByByte(bytes[4]);

          decode.Node_type = "LHT65N-VIB";
          
          if (bytes.length === 5)
            return decode;
        }
      }
    }
  }
  else if (port === 14) {
    decode.Timestamp_utc = (bytes[0] << 24) | (bytes[1] << 16) | (bytes[2] << 8) | bytes[3];
    decode.Datetime_utc = formatTimestamp(decode.Timestamp_utc);
    decode.BatV = (bytes[4] << 8 | bytes[5]) / 1000;
    
    decode.Node_type = "LHT65N-VIB";
    
    if (bytes.length === 6)
      return decode;
  }
  else if(port==5)
  {

  	var sub_band;
    var sensor;
    
    if(bytes[0]==0x3F)
      sensor= "LHT65N-VIB";
      
	  var firm_ver= (bytes[1]&0x0f)+'.'+(bytes[2]>>4&0x0f)+'.'+(bytes[2]&0x0f);
	  
    if(bytes[3]==0x01)
        freq_band="EU868";

  	else if(bytes[3]==0x09)

  	else if(bytes[3]==0x0A)

  	else if(bytes[3]==0x0F)

  	else if(bytes[3]==0x0D)

        freq_band="MA869";
  	
    if(bytes[4]==0xff)

	  else
